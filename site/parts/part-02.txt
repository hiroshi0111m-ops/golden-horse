 const w=Number(winOdds);

function settleRaceBets(top3){
 if(raceState.settled)return {total:0,wins:0};
 raceState.settled=true;state.raceBets=state.raceBets||[];state.betHistory=state.betHistory||[];
 let total=0,wins=0;const topSet=[...top3].sort((a,b)=>a-b).join('-'),top2Set=[...top3.slice(0,2)].sort((a,b)=>a-b).join('-');
 for(const b of state.raceBets){
  const hs=b.horses||[b.horse];let win=false;
  if(b.betType==='単勝')win=hs[0]===top3[0];
  else if(b.betType==='複勝')win=top3.includes(hs[0]);
  else if(b.betType==='馬連')win=[...hs].sort((a,b)=>a-b).join('-')===top2Set;
  else if(b.betType==='ワイド')win=hs.every(n=>top3.includes(n));
  else if(b.betType==='3連複')win=[...hs].sort((a,b)=>a-b).join('-')===topSet;
  else if(b.betType==='3連単')win=hs[0]===top3[0]&&hs[1]===top3[1]&&hs[2]===top3[2];
  if(b.betType==='複勝'){const fixed=placeSettlementOdds(hs[0]-1,top3);if(fixed!==null){b.odds=fixed;b.oddsFinal=true;}}
  const stake=Math.max(0,Number(b.stake)||0),odds=Math.max(0,Number(b.odds)||0);
  b.status=win?'win':'lose';b.payout=win?Math.max(0,Math.round(stake*odds)):0;
  if(win&&b.payout>0){state.gold=Math.max(0,Number(state.gold)||0)+b.payout;total+=b.payout;wins++}
 }
 save();renderHome();renderBetSlips();return {total,wins};
}
